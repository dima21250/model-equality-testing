\begin{frame}{Motivation: Beyond Binary Detection}

\textbf{Existing capability:} Detect distribution changes with MMD
\begin{itemize}
  \item Character-level kernel (Hamming)
  \item Statistical power, rigorous p-values
  \item Answer: \emph{"Did the distribution change?"} \good
\end{itemize}

\vspace{1em}

\textbf{Goal:} Interpretability — understand \emph{how} distributions differ
\begin{itemize}
  \item Quantum-inspired metrics (trace distance, Von Neumann entropy)
  \item Semantic embeddings to capture meaning
  \item Answer: \emph{"What changed semantically?"}
\end{itemize}

\end{frame}

% Quantum metrics approach
\begin{frame}{Quantum-Inspired Approach}

\textbf{Core idea:} Represent distribution of LLM outputs as quantum state

\vspace{0.5em}

\textbf{From completions to density matrices:}
\begin{enumerate}
  \item Start with N completions from model (text strings)
  \item Embed each completion: $\mathbf{x}_i \in \mathbb{R}^{768}$ (using MPNet or EmbeddingGemma)
  \item Normalize: $\psi_i = \mathbf{x}_i / ||\mathbf{x}_i||$ (unit vectors on sphere)
  \item Construct Gram matrix: $G_{ij} = \langle \psi_i | \psi_j \rangle$ (similarity between samples)
  \item Density matrix: $\rho = \frac{1}{N} G$ (normalized to unit trace)
\end{enumerate}

\vspace{0.5em}

\textbf{What the density matrix captures:}
\begin{itemize}
  \item Diagonal: individual sample "energies"
  \item Off-diagonal: correlations/overlaps between samples
  \item Eigenvalues: principal semantic dimensions
  \item Trace = 1 (normalization), Hermitian, positive semi-definite
\end{itemize}

\end{frame}

% Why quantum
\begin{frame}{Why Quantum Formalism?}

\textbf{Classical probability} (single distribution):
\begin{itemize}
  \item Represented by probability vector: $\mathbf{p} = (p_1, p_2, \ldots, p_k)$
  \item Events are mutually exclusive (one outcome at a time)
  \item No notion of "superposition" or "interference"
\end{itemize}

\vspace{0.5em}

\textbf{Quantum probability} (density matrix formalism):
\begin{itemize}
  \item Represented by density matrix: $\rho \in \mathbb{C}^{d \times d}$
  \item Captures correlations and "coherence" between states
  \item Natural geometric structure (Hilbert space)
  \item Rich set of information-theoretic metrics
\end{itemize}

\vspace{0.5em}

\textbf{Analogy:} Classical treats samples as points; quantum treats them as overlapping waves

\vspace{0.5em}

\textbf{Hypothesis:} Semantic embeddings have wave-like structure (overlaps matter)
\begin{itemize}
  \item Quantum metrics might capture subtle correlations
  \item Classical p-values miss geometric relationships
\end{itemize}

\end{frame}

% What the metrics mean
\begin{frame}{What Do the Quantum Metrics Mean?}

\textbf{1. Trace Distance} $D(\rho_A, \rho_B) = \frac{1}{2}\text{Tr}|\rho_A - \rho_B|$
\begin{itemize}
  \item Range: [0, 1] — geometric distance in quantum state space
  \item 0 = identical, 1 = orthogonal (perfectly separable)
  \item \textbf{Intuition:} Maximum probability of distinguishing distributions
\end{itemize}

\vspace{0.3em}

\textbf{2. Von Neumann Entropy} $S(\rho) = -\text{Tr}(\rho \log \rho)$
\begin{itemize}
  \item Range: [0, $\log d$] — diversity/uncertainty measure
  \item Low = tight clustering, High = broad spread
  \item \textbf{Interpretability:} Does quantization reduce diversity?
\end{itemize}

\vspace{0.3em}

\textbf{3. Quantum Relative Entropy} $D(\rho_A || \rho_B) = \text{Tr}(\rho_A \log \rho_A - \rho_A \log \rho_B)$
\begin{itemize}
  \item Range: [0, $\infty$) — information divergence (asymmetric)
  \item Measures information gained using $\rho_A$ vs $\rho_B$
  \item \textbf{Interpretability:} Information shift from implementation change
\end{itemize}

\end{frame}

\begin{frame}{Appendix: What is the Trace?}

\textbf{Definition:} The trace of a matrix is the sum of its diagonal elements.

$$\text{Tr}(A) = \sum_{i} A_{ii} = A_{11} + A_{22} + A_{33} + \ldots$$

\vspace{0.5em}

\textbf{Example:}
$$A = \begin{pmatrix} 3 & 1 & 2 \\ 5 & 4 & 6 \\ 7 & 8 & 9 \end{pmatrix} \quad \Rightarrow \quad \text{Tr}(A) = 3 + 4 + 9 = 16$$

\vspace{0.5em}

\textbf{Key properties:}
\begin{itemize}
  \item \textbf{Cyclic}: $\text{Tr}(ABC) = \text{Tr}(CAB) = \text{Tr}(BCA)$
  \item \textbf{Basis-independent}: Same value regardless of coordinate system
  \item \textbf{Linear}: $\text{Tr}(A + B) = \text{Tr}(A) + \text{Tr}(B)$
  \item \textbf{Eigenvalue sum}: $\text{Tr}(A) = \sum_i \lambda_i$ (sum of eigenvalues)
\end{itemize}

\vspace{0.5em}

\textbf{Why this matters:} Trace converts matrix operations to interpretable scalars

\end{frame}

\begin{frame}{Appendix: Why Trace in Quantum Metrics?}

\textbf{1. Normalization} — Density matrix: $\rho = \frac{1}{N}\sum_i |\psi_i\rangle\langle\psi_i|$

Require $\text{Tr}(\rho) = 1$ (total probability = 1)

Makes all density matrices comparable (same scale)

\vspace{0.3em}

\textbf{2. Von Neumann Entropy} — $S(\rho) = -\text{Tr}(\rho \log \rho)$

If $\rho$ has eigenvalues $\lambda_1, \ldots, \lambda_d$, then:
$$S(\rho) = -\sum_{i} \lambda_i \log \lambda_i$$

Trace extracts entropy from eigenvalue spectrum

\begin{itemize}
  \item All $\lambda_i = 1/d$ $\Rightarrow$ maximum entropy (uniform)
  \item One $\lambda_i = 1$, rest = 0 $\Rightarrow$ minimum entropy (deterministic)
\end{itemize}

\vspace{0.3em}

\textbf{3. Trace Distance} — $D(\rho_A, \rho_B) = \frac{1}{2}\text{Tr}|\rho_A - \rho_B|$

$|\rho_A - \rho_B|$ = matrix absolute value (all eigenvalues positive)

Trace sums up total difference across all dimensions

Factor $\frac{1}{2}$ normalizes to [0, 1] range

\end{frame}

\begin{frame}{Appendix: Trace in Our Density Matrix Construction}

\textbf{From embeddings to density matrix:}

\begin{enumerate}
  \item Start: N completions $\to$ embeddings $\mathbf{x}_1, \ldots, \mathbf{x}_N \in \mathbb{R}^{768}$
  \item Normalize: $\psi_i = \mathbf{x}_i / ||\mathbf{x}_i||$ (unit vectors)
  \item Gram matrix (N $\times$ N): $G_{ij} = \langle \psi_i | \psi_j \rangle$
  \item Density matrix: $\rho = \frac{1}{N} G$
\end{enumerate}

\vspace{0.5em}

\textbf{Verifying normalization:}
$$\text{Tr}(\rho) = \frac{1}{N}\sum_{i=1}^{N} G_{ii} = \frac{1}{N}\sum_{i=1}^{N} \langle\psi_i|\psi_i\rangle = \frac{1}{N} \cdot N = 1 \quad \checkmark$$

Because $||\psi_i|| = 1$ (normalized), diagonal elements $G_{ii} = 1$

\vspace{0.5em}

\textbf{What trace gives us:}
\begin{itemize}
  \item Converts N $\times$ N matrix to single number
  \item Basis-independent (doesn't depend on coordinate choice)
  \item Connects to eigenvalues: $\text{Tr}(\rho) = \sum_i \lambda_i$
\end{itemize}

\end{frame}

\begin{frame}{Appendix: Why Trace is Essential}

\textbf{Five reasons trace appears everywhere in quantum metrics:}

\vspace{0.5em}

\begin{enumerate}
  \item \textbf{Basis-independent}
  \begin{itemize}
    \item Results don't depend on how we orient embedding space
    \item Geometric invariant — same in any coordinate system
  \end{itemize}

  \vspace{0.3em}

  \item \textbf{Scalar output}
  \begin{itemize}
    \item Converts high-dimensional matrix operations to interpretable numbers
    \item Can compare traces directly (unlike comparing matrices)
  \end{itemize}

  \vspace{0.3em}

  \item \textbf{Probability interpretation}
  \begin{itemize}
    \item $\text{Tr}(\rho) = 1$ ensures proper normalization
    \item Connects to total probability in quantum mechanics
  \end{itemize}

  \vspace{0.3em}

  \item \textbf{Eigenvalue extraction}
  \begin{itemize}
    \item Trace "reads out" information from eigenvalue spectrum
    \item $\text{Tr}(A) = \sum_i \lambda_i$ — fundamental relationship
  \end{itemize}

  \vspace{0.3em}

  \item \textbf{Geometric meaning}
  \begin{itemize}
    \item In quantum mechanics, trace gives observable quantities
    \item Natural way to aggregate information across dimensions
  \end{itemize}
\end{enumerate}

\end{frame}

\begin{frame}{Appendix: The Sample-Size Artifact We Discovered}

\textbf{Problem:} Gram matrix size grows with sample size

\vspace{0.5em}

Our density matrix: $\rho = \frac{1}{N} G$ where $G$ is N $\times$ N

\vspace{0.3em}

When computing trace distance $\frac{1}{2}\text{Tr}|\rho_A - \rho_B|$:
\begin{itemize}
  \item N = 100: Trace sums 100 eigenvalues
  \item N = 1000: Trace sums 1000 eigenvalues
\end{itemize}

\vspace{0.5em}

\textbf{The artifact:}

More samples $\to$ bigger matrix $\to$ more terms in trace sum $\to$ higher distance

\emph{Even for identical distributions!}

\vspace{0.5em}

\textbf{Example} (synthetic data, N(0, I) vs N(0, I)):

\begin{center}
\begin{tabular}{cc}
\toprule
\textbf{N} & \textbf{Trace Distance} \\
\midrule
50 & 0.146 \\
100 & 0.212 \\
500 & 0.467 \\
1000 & 0.636 \\
\bottomrule
\end{tabular}
\end{center}

\vspace{0.3em}

\textbf{The trace operation is fine} — the problem is using an N $\times$ N matrix

\end{frame}

\begin{frame}{Appendix: What is a Gram Matrix?}

\textbf{Definition:} Given vectors $\mathbf{v}_1, \ldots, \mathbf{v}_N$, the Gram matrix is:

$$G_{ij} = \langle \mathbf{v}_i, \mathbf{v}_j \rangle = \mathbf{v}_i^T \mathbf{v}_j$$

The (i,j)-th entry is the \emph{inner product} (dot product) between vectors $\mathbf{v}_i$ and $\mathbf{v}_j$

\vspace{0.5em}

\textbf{Example:} Three 2-D vectors

$$\mathbf{v}_1 = \begin{pmatrix} 1 \\ 0 \end{pmatrix}, \quad \mathbf{v}_2 = \begin{pmatrix} 0 \\ 1 \end{pmatrix}, \quad \mathbf{v}_3 = \begin{pmatrix} 1 \\ 1 \end{pmatrix}$$

$$G = \begin{pmatrix}
\langle \mathbf{v}_1, \mathbf{v}_1 \rangle & \langle \mathbf{v}_1, \mathbf{v}_2 \rangle & \langle \mathbf{v}_1, \mathbf{v}_3 \rangle \\
\langle \mathbf{v}_2, \mathbf{v}_1 \rangle & \langle \mathbf{v}_2, \mathbf{v}_2 \rangle & \langle \mathbf{v}_2, \mathbf{v}_3 \rangle \\
\langle \mathbf{v}_3, \mathbf{v}_1 \rangle & \langle \mathbf{v}_3, \mathbf{v}_2 \rangle & \langle \mathbf{v}_3, \mathbf{v}_3 \rangle
\end{pmatrix} = \begin{pmatrix}
1 & 0 & 1 \\
0 & 1 & 1 \\
1 & 1 & 2
\end{pmatrix}$$

\vspace{0.3em}

\textbf{Key insight:} Gram matrix encodes all pairwise similarities between vectors

\end{frame}

\begin{frame}{Appendix: Properties of Gram Matrices}

\textbf{1. Symmetric:} $G_{ij} = G_{ji}$
\begin{itemize}
  \item Inner product is commutative: $\langle \mathbf{v}_i, \mathbf{v}_j \rangle = \langle \mathbf{v}_j, \mathbf{v}_i \rangle$
  \item Mirror symmetry across diagonal
\end{itemize}

\vspace{0.3em}

\textbf{2. Positive semi-definite:} $\mathbf{x}^T G \mathbf{x} \geq 0$ for all vectors $\mathbf{x}$
\begin{itemize}
  \item All eigenvalues $\geq 0$
  \item Geometrically: represents valid geometric configuration
\end{itemize}

\vspace{0.3em}

\textbf{3. Diagonal elements:} $G_{ii} = \langle \mathbf{v}_i, \mathbf{v}_i \rangle = ||\mathbf{v}_i||^2$
\begin{itemize}
  \item Self-similarity = squared length
  \item If vectors normalized ($||\mathbf{v}_i|| = 1$), then $G_{ii} = 1$
\end{itemize}

\vspace{0.3em}

\textbf{4. Off-diagonal elements:} $G_{ij} = ||\mathbf{v}_i|| \cdot ||\mathbf{v}_j|| \cdot \cos(\theta_{ij})$
\begin{itemize}
  \item Encodes angle between vectors
  \item For unit vectors: $G_{ij} = \cos(\theta_{ij}) \in [-1, 1]$
  \item $G_{ij} = 1$ $\Rightarrow$ parallel, $G_{ij} = 0$ $\Rightarrow$ orthogonal, $G_{ij} = -1$ $\Rightarrow$ opposite
\end{itemize}

\end{frame}

\begin{frame}{Appendix: What Does the Gram Matrix Represent?}

\textbf{Geometric interpretation:} The Gram matrix is a "map of similarities"

\vspace{0.5em}

Given N vectors in high-dimensional space (e.g., 768-D embeddings):

\begin{itemize}
  \item \textbf{Diagonal:} How "long" each vector is
  \item \textbf{Off-diagonal:} How "similar" pairs of vectors are
\end{itemize}

\vspace{0.5em}

\textbf{Key insight:} G captures all pairwise geometric relationships

\vspace{0.3em}

\textbf{Example interpretation} (normalized vectors, $G_{ij} \in [-1, 1]$):

\begin{center}
\begin{tabular}{cc}
\toprule
\textbf{$G_{ij}$ value} & \textbf{Meaning} \\
\midrule
1.0 & Vectors identical/parallel \\
0.5 & Moderate similarity (60° angle) \\
0.0 & Orthogonal (perpendicular) \\
-0.5 & Moderate dissimilarity (120° angle) \\
-1.0 & Opposite directions (180° angle) \\
\bottomrule
\end{tabular}
\end{center}

\vspace{0.3em}

\textbf{In our context:} High $G_{ij}$ = completions $i$ and $j$ are semantically similar

\end{frame}

\begin{frame}{Appendix: Why Gram Matrix in Our Construction?}

\textbf{From embeddings to quantum state — why the Gram matrix is the bridge:}

\vspace{0.5em}

\textbf{Step 1:} N completions $\to$ embeddings $\mathbf{x}_1, \ldots, \mathbf{x}_N \in \mathbb{R}^{768}$

\textbf{Step 2:} Normalize to unit sphere: $\psi_i = \mathbf{x}_i / ||\mathbf{x}_i||$

\textbf{Step 3:} Gram matrix: $G_{ij} = \langle \psi_i | \psi_j \rangle$

\textbf{Step 4:} Density matrix: $\rho = \frac{1}{N} G$

\vspace{0.5em}

\textbf{Why this works:}

\begin{enumerate}
  \item \textbf{Captures correlations:} Not just individual samples, but how they relate
  \item \textbf{Dimensionality reduction:} 768-D vectors $\to$ N $\times$ N similarity matrix
  \begin{itemize}
    \item For N = 100: $100 \times 768$ data $\to$ $100 \times 100$ Gram matrix
    \item Compresses information about relationships
  \end{itemize}
  \item \textbf{Quantum connection:} $G = \sum_i |\psi_i\rangle\langle\psi_i|$ in operator form
  \begin{itemize}
    \item Natural representation as density operator
    \item $\rho = \frac{1}{N}G$ satisfies $\text{Tr}(\rho) = 1$ (quantum normalization)
  \end{itemize}
  \item \textbf{Kernel methods:} Gram matrix = kernel matrix
  \begin{itemize}
    \item Same mathematical object used in SVMs, kernel PCA, etc.
    \item Well-studied properties and theory
  \end{itemize}
\end{enumerate}

\end{frame}

\begin{frame}{Appendix: Gram Matrix — What Information Does It Capture?}

\textbf{Consider two scenarios:}

\vspace{0.5em}

\textbf{Scenario A: Diverse outputs} (many different completions)

$$G_A = \begin{pmatrix}
1 & 0.1 & 0.05 & \ldots \\
0.1 & 1 & 0.08 & \ldots \\
0.05 & 0.08 & 1 & \ldots \\
\vdots & \vdots & \vdots & \ddots
\end{pmatrix} \quad \text{(small off-diagonals)}$$

Completions mostly orthogonal $\Rightarrow$ high diversity

\vspace{0.5em}

\textbf{Scenario B: Repetitive outputs} (similar completions)

$$G_B = \begin{pmatrix}
1 & 0.9 & 0.85 & \ldots \\
0.9 & 1 & 0.88 & \ldots \\
0.85 & 0.88 & 1 & \ldots \\
\vdots & \vdots & \vdots & \ddots
\end{pmatrix} \quad \text{(large off-diagonals)}$$

Completions highly correlated $\Rightarrow$ low diversity

\vspace{0.5em}

\textbf{Quantum metrics read this structure:}
\begin{itemize}
  \item Von Neumann entropy: $S(G_A) > S(G_B)$ (more diverse $\to$ higher entropy)
  \item Eigenvalues: $G_A$ has many small eigenvalues, $G_B$ has one large eigenvalue
\end{itemize}

\end{frame}

\begin{frame}{Appendix: The N $\times$ N Problem}

\textbf{Beautiful theory, practical problem:}

\vspace{0.5em}

Gram matrix size = N $\times$ N where N = number of samples

\vspace{0.3em}

\begin{center}
\begin{tabular}{ccc}
\toprule
\textbf{N samples} & \textbf{Gram matrix size} & \textbf{Eigenvalues to sum} \\
\midrule
50 & $50 \times 50$ & 50 \\
100 & $100 \times 100$ & 100 \\
1000 & $1000 \times 1000$ & 1000 \\
\bottomrule
\end{tabular}
\end{center}

\vspace{0.5em}

\textbf{The artifact:}

More samples $\to$ bigger Gram matrix $\to$ more dimensions $\to$ trace sums more terms

\vspace{0.3em}

When comparing two identical distributions:
\begin{itemize}
  \item N = 100: Trace distance = 0.212 (summing over 100 eigenvalues)
  \item N = 1000: Trace distance = 0.636 (summing over 1000 eigenvalues)
\end{itemize}

\vspace{0.5em}

\textbf{The fix we tried:} PCA to reduce to fixed $k \times k$ matrix
\begin{itemize}
  \item Eliminated sample-size artifact \good
  \item But weak discrimination (7\% effect) \bad
  \item Embedding space lacks signal for our differences
\end{itemize}

\end{frame}

\begin{frame}{Appendix: Summary — Gram Matrix}

\textbf{What is a Gram matrix?} Matrix of all pairwise inner products

$$G_{ij} = \langle \mathbf{v}_i, \mathbf{v}_j \rangle$$

\vspace{0.5em}

\textbf{Properties:}
\begin{itemize}
  \item Symmetric, positive semi-definite
  \item Diagonal = squared lengths, off-diagonal = similarities
  \item For unit vectors: entries are cosines of angles
\end{itemize}

\vspace{0.5em}

\textbf{Why use it?}
\begin{itemize}
  \item Captures all pairwise relationships between samples
  \item Bridge from vectors to quantum density operator
  \item Natural representation for kernel methods
  \item Enables quantum information-theoretic analysis
\end{itemize}

\vspace{0.5em}

\textbf{The challenge:} N $\times$ N size creates sample-size artifacts

\vspace{0.3em}

\textbf{The lesson:} Gram matrices are powerful for encoding structure, but size must be controlled for statistical testing

\end{frame}
